# LTL: the polarity of the formula decides the question. Negated asks whether
# the property holds; positive asks whether it can be refuted on this prefix.
ltl2ba -O c -H '"vars.h"' -f '!G({pressed} -> F {charge > min})' > ltl_example.ba-2.c
esbmc ltl_example.c --ltl ltl_example.ba-2.c -DLTL_PREFIX_BOUND=10

ltl2ba -O c -H '"vars.h"' -f 'G({pressed} -> F {charge > min})' > ltl_example-refute.ba-2.c
esbmc ltl_example.c --ltl ltl_example-refute.ba-2.c -DLTL_PREFIX_BOUND=10
